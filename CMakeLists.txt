cmake_minimum_required(VERSION 3.22)
project(BO2ZOffline VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Build with -A Win32 for retail BO2")
endif()
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
set(WITH_LTM OFF CACHE BOOL "" FORCE)
set(WITH_TFM OFF CACHE BOOL "" FORCE)
set(WITH_GMP OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory(external/libtomcrypt EXCLUDE_FROM_ALL)
set(STORAGE src/offline_storage.cpp src/offline_protocol.cpp src/log.cpp)
set(BACKEND src/dw_backend.cpp src/offline_auth.cpp src/offline_services.cpp src/offline_session.cpp ${STORAGE})
add_library(bo2offlineop SHARED src/module.cpp src/module.def src/module.rc ${BACKEND})
set_target_properties(bo2offlineop PROPERTIES OUTPUT_NAME "BO2Z-Offline")
target_include_directories(bo2offlineop PRIVATE src)
target_link_libraries(bo2offlineop PRIVATE ws2_32 iphlpapi bcrypt libtomcrypt)

# BO2Z-Offline is launched explicitly. It does not impersonate DXGI, WINMM,
# audio, Steam, or any other game dependency.
add_executable(BO2ZOfflineLauncher WIN32 launcher/launcher.cpp launcher/injector.cpp launcher/status_overlay.cpp launcher/launcher.rc)
target_include_directories(BO2ZOfflineLauncher PRIVATE launcher src)
set_target_properties(BO2ZOfflineLauncher PROPERTIES OUTPUT_NAME "BO2Z-Offline-Launcher")
target_compile_definitions(BO2ZOfflineLauncher PRIVATE UNICODE _UNICODE)
target_link_libraries(BO2ZOfflineLauncher PRIVATE gdiplus ole32 shell32 comctl32 bcrypt)
set(Z external/zlib)
add_library(rank_zlib STATIC ${Z}/adler32.c ${Z}/crc32.c ${Z}/deflate.c ${Z}/trees.c
  ${Z}/zutil.c ${Z}/inflate.c ${Z}/inftrees.c ${Z}/inffast.c ${Z}/compress.c)
target_include_directories(rank_zlib PUBLIC ${Z})
target_link_libraries(bo2offlineop PRIVATE rank_zlib advapi32)
enable_testing()
add_executable(dw_backend_tests tests/dw_backend_tests.cpp ${BACKEND})
target_include_directories(dw_backend_tests PRIVATE src)
target_link_libraries(dw_backend_tests PRIVATE ws2_32 iphlpapi bcrypt libtomcrypt)
add_test(NAME backend COMMAND dw_backend_tests)
add_test(NAME native_contracts COMMAND dw_backend_tests --contracts)
add_test(NAME isolated_service_contracts COMMAND dw_backend_tests --isolated-contracts)
set_tests_properties(isolated_service_contracts PROPERTIES TIMEOUT 30)
add_executable(rank_import_tests tests/rank_import_tests.cpp src/rank_profile.cpp ${STORAGE})
target_include_directories(rank_import_tests PRIVATE src)
target_link_libraries(rank_import_tests PRIVATE rank_zlib bcrypt libtomcrypt)
add_test(NAME rank_safety COMMAND rank_import_tests)
set_tests_properties(backend native_contracts rank_safety PROPERTIES TIMEOUT 30)
add_executable(module_load_tests tests/module_load_tests.cpp)
add_dependencies(module_load_tests bo2offlineop)
add_test(NAME standalone_module COMMAND module_load_tests)
set_tests_properties(standalone_module PROPERTIES TIMEOUT 10)
add_library(hook_probe SHARED tests/hook_probe.cpp tests/hook_probe.def)
target_include_directories(hook_probe PRIVATE src)
add_executable(hook_bootstrap_tests tests/hook_bootstrap_tests.cpp launcher/injector.cpp)
target_include_directories(hook_bootstrap_tests PRIVATE launcher src)
add_dependencies(hook_bootstrap_tests hook_probe BO2ZOfflineLauncher)
add_test(NAME launcher_direct_loader COMMAND hook_bootstrap_tests)
set_tests_properties(launcher_direct_loader PROPERTIES TIMEOUT 10)
add_executable(progress_diagnostics_tests tests/progress_diagnostics_tests.cpp src/progress_diagnostics.cpp src/rank_profile.cpp ${STORAGE})
target_include_directories(progress_diagnostics_tests PRIVATE src)
target_compile_definitions(progress_diagnostics_tests PRIVATE OFFLINE_DIAGNOSTICS_TEST)
target_link_libraries(progress_diagnostics_tests PRIVATE rank_zlib bcrypt libtomcrypt advapi32)
add_test(NAME progress_diagnostics COMMAND progress_diagnostics_tests)
set_tests_properties(progress_diagnostics PROPERTIES TIMEOUT 15)
add_executable(isolated_profile_tests tests/isolated_profile_tests.cpp ${STORAGE})
target_include_directories(isolated_profile_tests PRIVATE src)
target_compile_definitions(isolated_profile_tests PRIVATE OFFLINE_STORAGE_TEST)
target_link_libraries(isolated_profile_tests PRIVATE bcrypt libtomcrypt)
add_test(NAME isolated_profiles COMMAND isolated_profile_tests)
set_tests_properties(isolated_profiles PROPERTIES TIMEOUT 30)
